function handleCSVUpload(e) { const file = e.target.files[0]; if (!file) return; const reader = new FileReader(); reader.onload = function(event) { try { const rawText = event.target.result; const lines = rawText.split(/\r?\n/).filter(l => l.trim().length > 0); let savedContacts = JSON.parse(localStorage.getItem('local_patient_directory') || '[]'); let count = 0; // Helper to correctly parse CSV lines handling quoted commas function parseCSVLine(text) { let row = []; let inQuotes = false; let currentField = ''; for (let i = 0; i < text.length; i++) { let char = text[i]; if (char === '"') { if (inQuotes && text[i+1] === '"') { currentField += '"'; i++; } else { inQuotes = !inQuotes; } } else if ((char === ',' || char === '\t' || char === ';') && !inQuotes) { row.push(currentField.trim().replace(/^"|"$/g, '')); currentField = ''; } else { currentField += char; } } row.push(currentField.trim().replace(/^"|"$/g, '')); return row; } for (const line of lines) { const cells = parseCSVLine(line); if (cells.length === 0) continue; let nameParts = []; let phone = ''; for (const cell of cells) { if (!cell) continue; const digits = cell.replace(/[^0-9]/g, ''); // Check if cell is a phone number if (digits.length >= 6 && !phone && (cell.includes('+') || digits.length >= 9)) { phone = sanitizePhoneNumber(cell); } else if (!cell.includes('@') && !/^\d+$/.test(cell)) { // Collect any non-email, non-pure-number text as part of the name nameParts.push(cell); } else if (!phone && digits.length >= 6) { phone = sanitizePhoneNumber(cell); } } let name = nameParts.join(' ').trim(); const lowerName = name.toLowerCase(); // Skip header rows if (lowerName.includes('name') && (lowerName.includes('given') || lowerName.includes('first') || lowerName.includes('full') || lowerName.includes('mobile'))) { continue; } if (['name', 'full name', 'title', 'phone', 'mobile phone', 'given name', 'family name'].includes(lowerName)) { continue; } if (name) { const exists = savedContacts.some(p => p.name.toLowerCase() === name.toLowerCase()); if (!exists) { savedContacts.push({ name, phone }); count++; } } } localStorage.setItem('local_patient_directory', JSON.stringify(savedContacts)); alert(`Success! Imported ${count} new contacts. Total directory size: ${savedContacts.length}.`); loadPatientDirectory(); } catch (err) { alert(`Error parsing file: ${err.message}`); } }; reader.readAsText(file); e.target.value = ''; }
