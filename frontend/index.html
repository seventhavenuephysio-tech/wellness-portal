async function fetchAndRenderBookings() { try { const response = await fetch(API_BASE_URL + "/api/bookings"); if (response.ok) { const bookings = await response.json(); console.log("Current bookings:", bookings); // Select selected date const currentDate = document.getElementById("diaryDate").value; // Reset all slot buttons first document.querySelectorAll(".slot button").forEach(btn => { btn.disabled = false; btn.innerText = "BOOK"; btn.style.backgroundColor = "#1a73e8"; }); // Mark booked slots as booked bookings.forEach(b => { if (b.date === currentDate) { // Find matching therapist card and time slot const cards = document.querySelectorAll(".therapist-card"); cards.forEach(card => { const therapistName = card.querySelector("h3").innerText.trim(); if (therapistName === b.therapist) { const slots = card.querySelectorAll(".slot"); slots.forEach(slot => { const timeText = slot.querySelector("span").innerText.trim(); if (timeText === b.time) { const btn = slot.querySelector("button"); btn.disabled = true; btn.innerText = `Booked (${b.patientName})`; btn.style.backgroundColor = "#888"; } }); } }); } }); } } catch (error) { console.error("Failed to fetch bookings:", error); } }
