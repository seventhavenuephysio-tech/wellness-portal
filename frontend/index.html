// Live Render Backend Endpoint const API_URL = 'https://wellness-portal-api.onrender.com'; async function syncDatabase() { const statusBadge = document.getElementById('status-badge'); if (statusBadge) statusBadge.innerText = 'Connecting... ⏳'; try { // Fetch bookings from live API endpoint const response = await fetch(`${API_URL}/api/bookings`); if (response.ok) { const data = await response.json(); console.log('Successfully connected to backend! Bookings:', data); if (statusBadge) { statusBadge.innerText = 'Backend Server Online 🟢'; statusBadge.classList.add('online'); statusBadge.style.backgroundColor = '#e6ffe6'; statusBadge.style.color = '#2e7d32'; } } else { throw new Error(`Server returned status ${response.status}`); } } catch (error) { console.error('Connection failed:', error); if (statusBadge) { statusBadge.innerText = 'Backend Server Offline 🔴'; statusBadge.classList.remove('online'); statusBadge.style.backgroundColor = '#ffe6e6'; statusBadge.style.color = '#d9534f'; } } } // Attach click event to Sync Database button document.addEventListener('DOMContentLoaded', () => { const syncBtn = document.getElementById('sync-btn'); if (syncBtn) { syncBtn.addEventListener('click', syncDatabase); } // Auto-run sync check when page loads syncDatabase(); });
