function handleContactImport(event) { const file = event.target.files[0]; if (!file) return; const reader = new FileReader(); reader.onload = function(e) { const text = e.target.result; const lines = text.split('\n'); let count = 0; const dir = getSavedDirectory(); lines.forEach(line => { // Split line by commas, handling quoted text safely const cols = line.split(/,(?=(?:(?:[^"]*"){2})*[^"]*$)/); if (cols.length > 1) { let name = ""; let phone = ""; // Search through all columns in the row to find a name and a phone number cols.forEach(col => { const clean = col.replace(/['"]+/g, '').trim(); // Check if it looks like a phone number (digits, plus sign, spaces, dashes) if (/^\+?[0-9\s\-\(\)]{7,15}$/.test(clean)) { phone = clean; } else if (clean.length > 1 && !name && isNaN(clean)) { name = clean; // Capture text that isn't a number as the name } }); if (name && phone) { dir[name.toLowerCase()] = phone; count++; } } }); localStorage.setItem("patient_phonebook", JSON.stringify(dir)); alert(`Successfully imported ${count} contacts!`); }; reader.readAsText(file); }
