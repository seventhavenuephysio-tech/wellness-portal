async function handleCSVUpload(e) { const file = e.target.files[0]; if (!file) return; const reader = new FileReader(); reader.onload = async (event) => { const text = event.target.result; const lines = text.split(/\r?\n/).map(l => l.trim()).filter(l => l.length > 0); if (lines.length < 2) return; // Clean headers for matching const rawHeaders = lines[0].split(',').map(h => h.trim().toLowerCase().replace(/[^a-z0-9]/g, '')); const isBookingFile = rawHeaders.some(h => h.includes('slot') || h.includes('practitioner')); if (!isBookingFile) { // --- PATIENT CONTACTS IMPORT (Google / iCloud / Phonebook CSV) --- let nameIdx = rawHeaders.findIndex(h => h === 'name' || h.includes('givenname') || h.includes('displayname') || h.includes('first')); if (nameIdx === -1) nameIdx = rawHeaders.findIndex(h => h.includes('name')); let phoneIdx = rawHeaders.findIndex(h => h.includes('phone1value') || h.includes('phone') || h.includes('mobile') || h.includes('contact')); if (phoneIdx === -1) phoneIdx = 1; // Fallback to second column const contacts = []; for (let i = 1; i < lines.length; i++) { const row = lines[i].split(',').map(cell => cell.trim().replace(/^"|"$/g, '')); const name = row[nameIdx]; const rawPhone = row[phoneIdx] || ''; if (name && name.length > 0) { contacts.push({ name: name, phone: sanitizePhoneNumber(rawPhone) }); } } if (contacts.length === 0) { alert('No valid contact names or phone numbers found in file.'); return; } try { const res = await fetch(`${API_URL}/contacts/bulk`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ contacts }) }); if (res.ok) { alert(`Success! Imported ${contacts.length} contacts into your auto-fill directory.`); loadPatientDirectory(); } else { alert('Failed to save imported contacts.'); } } catch (err) { console.error('Contact import error:', err); alert('Error connecting to backend server.'); } } else { // --- APPOINTMENT SCHEDULE IMPORT --- const bookings = parseCSV(text); if (bookings.length === 0) { alert('No valid bookings found. CSV headers needed: Practitioner, Slot, Patient Name, Phone Number, Date'); return; } try { const res = await fetch(`${API_URL}/bookings/bulk`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ bookings }) }); const data = await res.json(); if (res.ok) { alert(`Successfully imported ${data.count} booking(s)!`); loadSchedule(); loadPatientDirectory(); } else { alert(`Import failed: ${data.error || 'Server error'}`); } } catch (err) { console.error('CSV import failed:', err); alert('Error connecting to backend server.'); } } }; reader.readAsText(file); e.target.value = ''; }
