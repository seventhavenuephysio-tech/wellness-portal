async function submitBooking() { if (!currentBookingTarget) return; const name = document.getElementById("patientNameInput").value.trim(); const phone = document.getElementById("patientPhoneInput").value.trim(); if (!name) return alert("Please enter the patient's full name."); const { practitioner, slot, date } = currentBookingTarget; const booking = { practitioner, therapist: practitioner, slot, time: slot, name, phone, date }; bookingsList.push(booking); closeModal(); renderColumns(getFilteredTherapists()); showToast("Appointment confirmed for " + name); // --- WhatsApp Integration --- if (phone) { // Remove spaces, plus signs, and special characters const cleanPhone = phone.replace(/[^0-9]/g, ''); const msg = `Hi ${name}, your appointment at Seventh Avenue Physio with ${practitioner} is confirmed for ${date} at ${slot}.`; // Opens WhatsApp with the pre-filled message window.open(`https://wa.me/${cleanPhone}?text=${encodeURIComponent(msg)}`, '_blank'); } try { await fetch('/api/bookings', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(booking) }); } catch (err) { console.error(err); } }
