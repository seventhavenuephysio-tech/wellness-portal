async function handleCSVUpload(e) { const file = e.target.files[0]; if (!file) return; const reader = new FileReader(); reader.onload = async (event) => { const text = event.target.result; const lines = text.split(/\r?\n/).map(l => l.trim()).filter(l => l.length > 0); if (lines.length < 2) return; // Parse CSV line preserving quoted values const parseRow = (line) => line.split(/,(?=(?:(?:[^"]*"){2})*[^"]*$)/).map(c => c.trim().replace(/^"|"$/g, '')); const headers = parseRow(lines[0]).map(h => h.toLowerCase().replace(/[^a-z0-9]/g, '')); // Identify Name and Phone column positions let nameIdx = headers.findIndex(h => h === 'name' || h.includes('displayname') || h.includes('givenname') || h.includes('first') || h.includes('patient')); if (nameIdx === -1) nameIdx = 0; let phoneIdx = headers.findIndex(h => h.includes('phone') || h.includes('mobile') || h.includes('value') || h.includes('cell') || h.includes('number')); if (phoneIdx === -1) phoneIdx = 1; const contacts = []; for (let i = 1; i < lines.length; i++) { const row = parseRow(lines[i]); const name = row[nameIdx] || row[0] || ''; const rawPhone = row[phoneIdx] || row[1] || ''; if (name && name.length > 0) { contacts.push({ practitioner: 'Chido', slot: '08:00 AM', name: name, phone: sanitizePhoneNumber(rawPhone), date: '2000-01-01' }); } } if (contacts.length === 0) { alert('No valid contacts found in CSV file.'); return; } try { const res = await fetch(`${API_URL}/bookings/bulk`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ bookings: contacts }) }); if (res.ok) { alert(`Success! Imported ${contacts.length} patient phone numbers into your directory.`); loadPatientDirectory(); } else { alert('Failed to save imported contacts.'); } } catch (err) { console.error('Contact import error:', err); alert('Error connecting to backend server.'); } }; reader.readAsText(file); e.target.value = ''; }
