function handleContactImport(event) { const file = event.target.files[0]; if (!file) return; const reader = new FileReader(); reader.onload = function(e) { const text = e.target.result; const lines = text.split('\n').filter(line => line.trim() !== ''); if (lines.length < 2) return alert("Invalid CSV file."); const parseCSVLine = (line) => { const values = []; let current = ''; let inQuotes = false; for (let i = 0; i < line.length; i++) { const char = line[i]; if (char === '"') { inQuotes = !inQuotes; } else if (char === ',' && !inQuotes) { values.push(current.trim()); current = ''; } else { current += char; } } values.push(current.trim()); return values; }; const headers = parseCSVLine(lines[0]).map(h => h.replace(/['"]+/g, '').toLowerCase()); // Identify column indexes for Names and Phone Numbers const nameIndexes = []; const phoneIndexes = []; headers.forEach((h, idx) => { if (h.includes('name') || h.includes('given') || h.includes('family') || h.includes('fn')) { nameIndexes.push(idx); } if (h.includes('phone') || h.includes('mobile') || h.includes('cell') || h.includes('tel')) { phoneIndexes.push(idx); } }); const dir = getSavedDirectory(); let count = 0; for (let i = 1; i < lines.length; i++) { const cols = parseCSVLine(lines[i]); if (!cols.length) continue; // Assemble full name from relevant name columns let nameParts = []; if (nameIndexes.length > 0) { nameIndexes.forEach(idx => { if (cols[idx]) { const clean = cols[idx].replace(/['"]+/g, '').trim(); if (clean && !clean.toLowerCase().includes('http')) nameParts.push(clean); } }); } let fullName = nameParts.join(' ').trim(); // Find first valid phone number across all phone columns or remaining columns let phone = ''; if (phoneIndexes.length > 0) { for (let idx of phoneIndexes) { if (cols[idx]) { const clean = cols[idx].replace(/['"]+/g, '').trim(); const digits = clean.replace(/[^0-9+]/g, ''); if (digits.length >= 7) { phone = clean; break; } } } } // Fallback scan across all columns if standard headers were not found if (!fullName || !phone) { cols.forEach(col => { const clean = col.replace(/['"]+/g, '').trim(); const digits = clean.replace(/[^0-9+]/g, ''); if (digits.length >= 7 && !phone) { phone = clean; } else if (clean.length > 1 && !fullName && isNaN(clean)) { fullName = clean; } }); } if (fullName && phone) { dir[fullName.toLowerCase()] = phone; count++; } } localStorage.setItem("patient_phonebook", JSON.stringify(dir)); alert(`Successfully imported ${count} complete patient contacts!`); updatePatientDatalist(); }; reader.readAsText(file); }
