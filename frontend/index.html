const API_URL = 'https://wellness-portal-api.onrender.com'; async function syncDatabase() { const statusBadge = document.getElementById('status-badge'); if (statusBadge) statusBadge.innerText = 'Connecting... ⏳'; try { // Fetch the root API endpoint const response = await fetch(`${API_URL}/`); if (response.ok) { const data = await response.json(); console.log('Connected to backend:', data); if (statusBadge) { statusBadge.innerText = 'Backend Server Online 🟢'; statusBadge.style.backgroundColor = '#e6ffe6'; statusBadge.style.color = '#2e7d32'; } } else { throw new Error(`Server returned status ${response.status}`); } } catch (error) { console.error('Connection failed:', error); if (statusBadge) { statusBadge.innerText = 'Backend Server Offline 🔴'; statusBadge.style.backgroundColor = '#ffe6e6'; statusBadge.style.color = '#d9534f'; } } } // Check status on click and on page load document.addEventListener('DOMContentLoaded', syncDatabase);
